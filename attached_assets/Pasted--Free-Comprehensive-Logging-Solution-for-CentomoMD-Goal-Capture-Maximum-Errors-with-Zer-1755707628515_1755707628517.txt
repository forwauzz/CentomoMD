## 📁 **Layer 2: Server-Side File Logging**

### **Replit Logging Strategy:**
Since Replit doesn't persist files, we'll use multiple approaches:

#### **1. Console Logging (Always Available)**
```javascript
// Structured console logs that show in Replit console
const serverLog = (level, category, event, data) => {
  const logEntry = {
    timestamp: new Date().toISOString(),
    level,
    category,
    event,
    data,
    pid: process.pid,
    memory: process.memoryUsage()
  };
  
  console.log(`[${level}] ${category}:${event}`, JSON.stringify(logEntry));
};
```

#### **2. In-Memory Log Buffer**
```javascript
// Keep recent logs in memory (survives short restarts)
class MemoryLogBuffer {
  constructor(maxSize = 1000) {
    this.logs = [];
    this.maxSize = maxSize;
  }
  
  add(logEntry) {
    this.logs.push({
      ...logEntry,
      id: Date.now() + Math.random()
    });
    
    if (this.logs.length > this.maxSize) {
      this.logs.shift(); // Remove oldest
    }
  }
  
  getLogs(filter = {}) {
    return this.logs.filter(log => {
      if (filter.level && log.level !== filter.level) return false;
      if (filter.category && log.category !== filter.category) return false;
      if (filter.since && new Date(log.timestamp) < filter.since) return false;
      return true;
    });
  }
}

const logBuffer = new MemoryLogBuffer(2000);
```

#### **3. External Storage Backup**
```javascript
// Send critical logs to free external storage
const backupCriticalLogs = async (logs) => {
  // Option 1: GitHub Gist (free, private)
  // Option 2: Google Drive API (free tier)
  // Option 3: Webhook to free service (Zapier, IFTTT)
  
  try {
    await fetch('https://api.github.com/gists', {
      method: 'POST',
      headers: {
        'Authorization': `token ${process.env.GITHUB_TOKEN}`,
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({
        description: `CentomoMD Logs ${new Date().toISOString()}`,
        public: false,
        files: {
          'logs.json': {
            content: JSON.stringify(logs, null, 2)
          }
        }
      })
    });
  } catch (error) {
    console.error('Failed to backup logs:', error);
  }
};
```

## 🔄 **Layer 3: Session Recording & State Tracking**

### **User Journey Tracking:**
```javascript
class SessionTracker {
  constructor() {
    this.sessionId = this.generateSessionId();
    this.startTime = Date.now();
    this.events = [];
    this.state = {};
  }
  
  trackEvent(type, data) {
    const event = {
      timestamp: Date.now(),
      sessionTime: Date.now() - this.startTime,
      type,
      data: this.sanitizeData(data),
      url: window.location.pathname,
      referrer: document.referrer
    };
    
    this.events.push(event);
    this.saveToStorage();
    
    // Send to server if critical
    if (this.isCriticalEvent(type)) {
      this.sendToServer(event);
    }
  }
  
  trackStateChange(component, newState) {
    this.state[component] = {
      ...newState,
      timestamp: Date.now()
    };
    this.saveToStorage();
  }
  
  sanitizeData(data) {
    // Remove any potential medical data
    const sanitized = { ...data };
    delete sanitized.content;
    delete sanitized.transcript;
    delete sanitized.formData;
    
    // Keep only metadata
    return {
      hasContent: !!data.content,
      contentLength: data.content?.length || 0,
      contentType: typeof data.content,
      fieldCount: data.formData ? Object.keys(data.formData).length : 0,
      ...sanitized
    };
  }
}
```

## 🚨 **Layer 4: Comprehensive Error Handling**

### **Global Error Capture:**
```javascript
// Capture ALL possible errors
class ComprehensiveErrorHandler {
  constructor() {
    this.setupGlobalHandlers();
    this.setupNetworkMonitoring();
    this.setupPerformanceMonitoring();
  }
  
  setupGlobalHandlers() {
    // Unhandled JavaScript errors
    window.addEventListener('error', (event) => {
      this.logError('JAVASCRIPT_ERROR', {
        message: event.message,
        filename: event.filename,
        lineno: event.lineno,
        colno: event.colno,
        stack: event.error?.stack,
        url: window.location.href
      });
    });
    
    // Unhandled promise rejections
    window.addEventListener('unhandledrejection', (event) => {
      this.logError('PROMISE_REJECTION', {
        reason: event.reason?.message || event.reason,
        stack: event.reason?.stack,
        url: window.location.href
      });
    });
    
    // React error boundary integration
    this.setupReactErrorBoundary();
  }
  
  setupNetworkMonitoring() {
    // Monitor all fetch requests
    const originalFetch = window.fetch;
    window.fetch = async (...args) => {
      const startTime = Date.now();
      const [url, options] = args;
      
      try {
        const response = await originalFetch(...args);
        
        // Log successful requests
        this.logEvent('API_REQUEST_SUCCESS', {
          url: url.toString(),
          method: options?.method || 'GET',
          status: response.status,
          duration: Date.now() - startTime
        });
        
        return response;
      } catch (error) {
        // Log failed requests
        this.logError('API_REQUEST_FAILED', {
          url: url.toString(),
          method: options?.method || 'GET',
          error: error.message,
          duration: Date.now() - startTime
        });
        
        throw error;
      }
    };
  }
  
  setupPerformanceMonitoring() {
    // Monitor page load performance
    window.addEventListener('load', () => {
      setTimeout(() => {
        const navigation = performance.getEntriesByType('navigation')[0];
        this.logEvent('PAGE_PERFORMANCE', {
          loadTime: navigation.loadEventEnd - navigation.loadEventStart,
          domContentLoaded: navigation.domContentLoadedEventEnd - navigation.domContentLoadedEventStart,
          firstPaint: performance.getEntriesByName('first-paint')[0]?.startTime,
          memoryUsage: (performance as any).memory ? {
            used: (performance as any).memory.usedJSHeapSize,
            total: (performance as any).memory.totalJSHeapSize,
            limit: (performance as any).memory.jsHeapSizeLimit
          } : null
        });
      }, 1000);
    });
  }
}
```

## 📊 **Layer 5: Local Log Dashboard**

### **Simple Admin Interface:**
Create a hidden admin page (`/admin/logs`) that shows all captured data:

```javascript
// React component for viewing logs
const LogDashboard = () => {
  const [logs, setLogs] = useState([]);
  const [filter, setFilter] = useState({ level: 'all', category: 'all' });
  
  useEffect(() => {
    loadLogs();
  }, [filter]);
  
  const loadLogs = async () => {
    // Load from localStorage
    const clientLogs = JSON.parse(localStorage.getItem('centomoLogs') || '[]');
    
    // Load from server
    const serverLogs = await fetch('/api/logs/recent').then(r => r.json());
    
    // Combine and sort
    const allLogs = [...clientLogs, ...serverLogs]
      .sort((a, b) => new Date(b.timestamp) - new Date(a.timestamp));
    
    setLogs(allLogs);
  };
  
  return (
    <div className="log-dashboard">
      <h2>CentomoMD Error Logs</h2>
      
      {/* Filters */}
      <div className="filters">
        <select onChange={(e) => setFilter({...filter, level: e.target.value})}>
          <option value="all">All Levels</option>
          <option value="ERROR">Errors Only</option>
          <option value="WARN">Warnings</option>
          <option value="INFO">Info</option>
        </select>
      </div>
      
      {/* Log entries */}
      <div className="log-entries">
        {logs.map(log => (
          <LogEntry key={log.id} log={log} />
        ))}
      </div>
      
      {/* Export options */}
      <div className="export-options">
        <button onClick={() => downloadLogs(logs)}>Download All Logs</button>
        <button onClick={() => clearOldLogs()}>Clear Old Logs</button>
      </div>
    </div>
  );
};
```
