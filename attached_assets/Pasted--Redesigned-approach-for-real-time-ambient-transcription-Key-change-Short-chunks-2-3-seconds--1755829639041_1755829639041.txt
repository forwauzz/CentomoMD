class StreamingAmbientProcessor {
private config = {
chunkDurationMs: 3000,        // 3 seconds instead of 30
overlapMs: 500,               // 0.5 second overlap
maxConcurrentChunks: 2,       // Process max 2 chunks simultaneously
silenceThreshold: 0.01,       // VAD threshold
minChunkDuration: 1000        // Minimum 1 second before processing
};

private activeChunks = new Map<string, AudioChunk>();
private transcriptionBuffer: TranscriptionSegment[] = [];
private isRecording = false;
private currentRecorder: MediaRecorder | null = null;

async startAmbientListening(): Promise<void> {
const stream = await navigator.mediaDevices.getUserMedia({ audio: true });

```
// Use timeslice for regular chunk generation
this.currentRecorder = new MediaRecorder(stream, {
  mimeType: 'audio/webm;codecs=opus'
});

// Key change: Use timeslice for automatic chunking
this.currentRecorder.start(this.config.chunkDurationMs); // 3-second chunks

this.currentRecorder.ondataavailable = (event) => {
  if (event.data.size > 0) {
    this.processAudioChunk(event.data);
  }
};

this.isRecording = true;
```

}

private async processAudioChunk(audioBlob: Blob): Promise<void> {
const chunkId = `chunk-${Date.now()}`;

```
// Quick VAD check to avoid processing silence
if (await this.containsSpeech(audioBlob)) {
  this.activeChunks.set(chunkId, {
    id: chunkId,
    data: audioBlob,
    timestamp: Date.now()
  });

  // Process immediately - don't wait
  this.transcribeChunk(chunkId, audioBlob);
}
```

}

private async transcribeChunk(chunkId: string, audioBlob: Blob): Promise<void> {
try {
// Try browser Whisper first (should be fast for 3-second chunks)
const result = await this.browserWhisper.transcribe(audioBlob);

```
  if (result.text.trim()) {
    this.addTranscriptionSegment({
      id: chunkId,
      text: result.text,
      timestamp: Date.now(),
      source: 'browser-whisper',
      confidence: result.confidence || 0.8
    });
  }
} catch (error) {
  console.log(`Browser Whisper failed for ${chunkId}, using server fallback`);
  // Server fallback for failed chunks
  this.transcribeChunkOnServer(chunkId, audioBlob);
} finally {
  this.activeChunks.delete(chunkId);
}
```

}

private addTranscriptionSegment(segment: TranscriptionSegment): void {
this.transcriptionBuffer.push(segment);

```
// Immediately update UI - this is the key difference
this.callbacks.onTranscriptionUpdate?.(segment);

// Clean old segments (keep last 50 for context)
if (this.transcriptionBuffer.length > 50) {
  this.transcriptionBuffer = this.transcriptionBuffer.slice(-50);
}
```

}

private async containsSpeech(audioBlob: Blob): Promise<boolean> {
// Simple VAD - you could use WebRTC VAD or energy-based detection
// For now, just check if the blob has reasonable size
return audioBlob.size > 1000; // Rough threshold
}
}

}

useEffect(() => {

.map(segment => {
const time = new Date(segment.timestamp).toLocaleTimeString();
const sourceIcon = segment.source === ‘browser-whisper’ ? ‘🔵’ : ‘🔴’;
return `[${time}] ${sourceIcon} ${segment.text}`;
})
.join(’\n’);

```

```

}, [segments]);

return (
