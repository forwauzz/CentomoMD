#### **3. Audio Conversion Utility**
**File:** `server/audio-convert.ts` (CREATE NEW FILE)

```typescript
import ffmpeg from "fluent-ffmpeg";
import ffmpegPath from "@ffmpeg-installer/ffmpeg";
import { Readable } from "stream";

ffmpeg.setFfmpegPath(ffmpegPath.path);

export function webmToWavMono16k(input: Buffer): Promise<Buffer> {
  return new Promise((resolve, reject) => {
    const inStream = Readable.from(input);
    const chunks: Buffer[] = [];
    
    const out = ffmpeg(inStream)
      .inputFormat("webm")
      .noVideo()
      .audioChannels(1)
      .audioFrequency(16000)
      .format("wav")
      .on("error", reject)
      .pipe();
      
    out.on("data", (d: Buffer) => chunks.push(d));
    out.on("end", () => resolve(Buffer.concat(chunks)));
  });
}
```

#### **4. Updated Whisper Service**
**File:** `server/whisper-service.ts` (MODIFY EXISTING)

```typescript
import OpenAI, { toFile } from "openai";
import { webmToWavMono16k } from "./audio-convert";

const openai = new OpenAI({ apiKey: process.env.OPENAI_API_KEY });

export async function transcribeAudioWithWhisper(args: {
  buffer: Buffer;
  filename: string;
  mimetype?: string;
  language?: "en" | "fr" | "auto";
  temperature?: number;
  sessionId?: string;
  chunkIndex?: number;
  mode: "transcribe" | "other";
}) {
  const { buffer, filename, mimetype, language, temperature = 0.2 } = args;

  // 1) Try direct WebM upload (fast path)
  try {
    const file = await toFile(buffer, filename, { type: mimetype ?? "audio/webm" });
    return await openai.audio.transcriptions.create({
      model: "whisper-1", // Use whisper-1 for stability
      file,
      ...(language && language !== "auto" ? { language } : {}),
      temperature,
      response_format: "json",
    });
  } catch (e: any) {
    // Only fall back on format/decoder problems
    const m = (e?.message || "").toLowerCase();
    if (!m.includes("decoded") && !m.includes("format")) throw e;
    
    console.warn("[whisper] WebM failed, falling back to WAV conversion");
  }

  // 2) Fallback: convert in-memory to 16k mono WAV (no disk)
  const wav = await webmToWavMono16k(buffer);
  const wavFile = await toFile(wav, filename.replace(/\.webm$/i, ".wav"), { type: "audio/wav" });
  
  return await openai.audio.transcriptions.create({
    model: "whisper-1",
    file: wavFile,
    ...(language && language !== "auto" ? { language } : {}),
    temperature,
    response_format: "json",
  });
}
```

#### **6. Updated API Endpoint**
**File:** `server/routes.ts` (MODIFY EXISTING `/api/transcribe-ambient-chunk`)

```typescript
import multer from "multer";
import helmet from "helmet";
import { audioLimiter } from "./security";

// Add helmet middleware (if not already present)
app.use(helmet({ contentSecurityPolicy: false }));

const upload = multer({
  storage: multer.memoryStorage(),
  limits: { fileSize: 8 * 1024 * 1024 }, // 8MB safety limit
  fileFilter: (_req, file, cb) =>
    /audio\/(webm|wav|mp3|m4a|aac|ogg|flac)/.test(file.mimetype)
      ? cb(null, true) 
      : cb(new Error("UNSUPPORTED_AUDIO_TYPE")),
});

app.post("/api/transcribe-ambient-chunk", 
  audioLimiter, 
  upload.single("file"), 
  async (req, res) => {
    try {
      if (!req.file?.buffer?.length) {
        return res.status(400).json({ error: "NO_FILE" });
      }

      const { sessionId, chunkIndex, language, mode } = req.body;

      // TODO: Add consent verification here
      // const hasConsent = await verifyExternalProcessingConsent(userId);
      // if (!hasConsent) return res.status(403).json({ error: "CONSENT_REQUIRED" });

      const result = await transcribeAudioWithWhisper({
        buffer: req.file.buffer,
        filename: req.file.originalname || `chunk-${Date.now()}.webm`,
        mimetype: req.file.mimetype,
        language,
        temperature: 0.2,
        sessionId,
        chunkIndex: Number(chunkIndex ?? 0),
        mode,
      });

      // Audit logging (metadata only, no PHI)
      console.info("[ambient] transcription completed", {
        sessionId,
        chunkIndex: Number(chunkIndex ?? 0),
        audioSize: req.file.size,
        success: true,
        provider: "openai"
      });

      // Return text only; do NOT store
      return res.json({ 
        text: result.text, 
        chunkIndex: Number(chunkIndex ?? 0) 
      });

    } catch (e: any) {
      // No PHI in logs
      console.warn("[ambient] chunk failed", { 
        code: e.code, 
        msg: String(e.message).slice(0, 120) 
      });
      return res.status(500).json({ error: "TRANSCRIPTION_FAILED" });
    } finally {
      // Defensive cleanup (zero-retention)
      if (global.gc) {
        try { 
          global.gc(); 
        } catch {}
      }
    }
  }
);
```

## 🎯 **PHASE 2: LOCAL WHISPER IMPLEMENTATION PLAN**

### **📝 Implementation Documentation for Future Reference**

#### **Why Local Whisper?**
- **True Zero-Retention Compliance:** No patient data leaves our servers
- **Cost Reduction:** No per-minute OpenAI API costs
- **Privacy Assurance:** Complete control over patient data
- **Performance:** Potential for faster processing with dedicated hardware
- **Regulatory Compliance:** Eliminates external data processing concerns

#### **Technical Architecture for Local Whisper**

##### **Option 1: faster-whisper (Recommended)**
```typescript
// Future implementation structure
import { FasterWhisper } from "faster-whisper-node";

class LocalWhisperService {
  private model: FasterWhisper;
  
  constructor() {
    this.model = new FasterWhisper({
      modelPath: "/models/whisper-medium", // Downloaded model weights
      device: "cpu", // or "cuda" for GPU acceleration
      compute_type: "int8" // Optimized for CPU
    });
  }
  
  async transcribe(audioBuffer: Buffer, language?: string): Promise<{ text: string }> {
    // Convert WebM to WAV in-memory (reuse existing webmToWavMono16k)
    const wavBuffer = await webmToWavMono16k(audioBuffer);
    
    // Transcribe locally - no network calls
    const result = await this.model.transcribe(wavBuffer, {
      language: language === "auto" ? undefined : language,
      task: "transcribe"
    });
    
    return { text: result.text };
  }
}
```

##### **Option 2: whisper.cpp via Node.js bindings**
```typescript
// Alternative implementation
import { whisper } from "whisper-node";

class WhisperCppService {
  async transcribe(audioBuffer: Buffer, language?: string): Promise<{ text: string }> {
    const wavBuffer = await webmToWavMono16k(audioBuffer);
    
    // Use whisper.cpp for local transcription
    const result = await whisper(wavBuffer, {
      modelName: "medium",
      language: language === "auto" ? undefined : language,
      format: "json"
    });
    
    return { text: result.transcription };
  }
}
```
